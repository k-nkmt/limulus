data out;
  set inp;
  y_propcase = propcase(name);
  y_cat = cat(name, ' test');
  y_cats = cats(' a ', ' b ');
  y_catt = catt('hello ', ' world');
  y_catx = catx('-', 'a', 'b', 'c');
  y_index = index(name, 'world');
  y_find = find(name, 'WORLD', 1, 'i');
  y_tranwrd = tranwrd(name, 'hello', 'hi');
  y_translate = translate(name, 'HW', 'hw');
  y_length = length(name);
  y_lengthn = lengthn(name);
  y_strip = strip(name);
  y_reverse = reverse(name);
  y_repeat = repeat('ab', 3);
  y_countw = countw(name);
  keep y_propcase y_cat y_cats y_catt y_catx y_index y_find y_tranwrd y_translate y_length y_lengthn y_strip y_reverse y_repeat y_countw;
run;
